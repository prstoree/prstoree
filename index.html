Integrasi Google Apps Script Backend Webhook

Gunakan berkas code.gs untuk mengaktifkan sinkronisasi dua arah secara langsung dengan Google Sheets.

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    sheet.clear();
    for (var i = 0; i < contents.rows.length; i++) {
      sheet.appendRow(contents.rows[i]);
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
           .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
           .setMimeType(ContentService.MimeType.JSON);
  }
}
